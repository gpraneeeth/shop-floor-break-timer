function simpleHash(str) {
let hash = 5381;
for (let i = 0; i < str.length; i++) {
hash = ((hash << 5) + hash) + str.charCodeAt(i);
hash = hash & hash;
}
return hash.toString(36);
}

function handleAuthSubmit() {
const pw = passwordInput.value;
if (mode === 'setup') {
const confirmPw = passwordConfirm.value;

loginError.textContent = 'Password must be at least 4 characters.';

if (pw !== confirmPw) {

unlockApp();
} else {

} else {
loginError.textContent = 'Incorrect password.';
}
}
}

const GRACE_SECONDS = 5 * 60;
const breakConfig = {
break1: { label: 'Break 1', minutes: 15 },
lunch: { label: 'Lunch Break', minutes: 30 },
break2: { label: 'Break 2', minutes: 15 }
};

let selectedBreak = 'break1';
let activeBreaks = [];
let masterInterval = null;

const nameInput = document.getElementById('nameInput');
const breakRow = document.getElementById('breakRow');
const startBtn = document.getElementById('startBtn');
const activeContainer = document.getElementById('activeContainer');
const logContainer = document.getElementById('logContainer');
const clearLogBtn = document.getElementById('clearLogBtn');

function formatTime(totalSeconds) {
const s = Math.max(0, Math.round(totalSeconds));
const m = Math.floor(s / 60).toString().padStart(2, '0');
const sec = (s % 60).toString().padStart(2, '0');
return `${m}:${sec}`;
}

function playChime(freq) {
try {
const ctx = new (window.AudioContext || window.webkitAudioContext)();
const osc = ctx.createOscillator();
const gain = ctx.createGain();
osc.connect(gain);
gain.connect(ctx.destination);
osc.frequency.value = freq;
gain.gain.setValueAtTime(0.15, ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.7);
osc.start();
osc.stop(ctx.currentTime + 0.7);
} catch (e) {}
}

function selectBreak(key) {
selectedBreak = key;
document.querySelectorAll('.break-btn').forEach(b => {
b.classList.toggle('selected', b.dataset.break === key);
});
}

function getPhaseInfo(breakType, elapsedSeconds) {
const allotted = breakConfig[breakType].minutes * 60;
const graceEnd = allotted + GRACE_SECONDS;
if (elapsedSeconds <= allotted) {
return { phase: 'normal', displayText: formatTime(allotted - elapsedSeconds), label: 'Remaining' };
} else if (elapsedSeconds <= graceEnd) {
return { phase: 'grace', displayText: formatTime(graceEnd - elapsedSeconds), label: 'Grace Left' };
} else {
return { phase: 'overtime', displayText: '+' + formatTime(elapsedSeconds - graceEnd), label: 'Overtime' };
}
}

function startBreak() {
const name = nameInput.value.trim();
if (!name) {
nameInput.focus();
nameInput.style.borderColor = '#ef4444';
setTimeout(() => nameInput.style.borderColor = '#334155', 1200);
return;
}
activeBreaks.push({
id: Date.now() + '-' + Math.random().toString(36).slice(2, 7),
name,
breakType: selectedBreak,
elapsedSeconds: 0,
lastPhase: 'normal'
});
nameInput.value = '';
nameInput.focus();
renderActive();
ensureTicking();
}

function stopBreak(id) {
const idx = activeBreaks.findIndex(b => b.id === id);
if (idx === -1) return;
const entry = activeBreaks[idx];
const allotted = breakConfig[entry.breakType].minutes * 60;
const graceEnd = allotted + GRACE_SECONDS;
let status, statusClass;

if (entry.elapsedSeconds <= allotted) {
status = 'On time'; statusClass = 'status-ok';
} else if (entry.elapsedSeconds <= graceEnd) {
status = 'Used grace period'; statusClass = 'status-grace';
} else {
status = 'Late'; statusClass = 'status-late';
}

addLogEntry({
name: entry.name,
breakLabel: breakConfig[entry.breakType].label,
actual: formatTime(entry.elapsedSeconds),
status, statusClass,
time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
});

activeBreaks.splice(idx, 1);
renderActive();
}

function ensureTicking() {
if (masterInterval) return;
masterInterval = setInterval(() => {
activeBreaks.forEach(entry => {
entry.elapsedSeconds++;
const { phase } = getPhaseInfo(entry.breakType, entry.elapsedSeconds);
if (phase !== entry.lastPhase) {
if (phase === 'grace') playChime(660);
if (phase === 'overtime') playChime(330);
entry.lastPhase = phase;
}
});
renderActive();
if (activeBreaks.length === 0) {
clearInterval(masterInterval);
masterInterval = null;
}
}, 1000);
}

function renderActive() {
if (activeBreaks.length === 0) {
activeContainer.innerHTML = '<div class="empty-log">No one is currently on break</div>';
return;
}
let html = '';
activeBreaks.forEach(entry => {
const { phase, displayText, label } = getPhaseInfo(entry.breakType, entry.elapsedSeconds);
html += `<div class="person-card">
<div class="person-info">
<div class="person-name">${escapeHtml(entry.name)}</div>
<div class="person-break">${breakConfig[entry.breakType].label}</div>
</div>
<div class="person-timer">
<div class="person-time ${phase}">${displayText}</div>
<div class="person-phase">${label}</div>
</div>
<button class="person-stop" data-id="${entry.id}">Stop</button>
</div>`;
});
activeContainer.innerHTML = html;
activeContainer.querySelectorAll('.person-stop').forEach(btn => {
btn.onclick = () => stopBreak(btn.dataset.id);
});
}

function loadLog() {
try {
return JSON.parse(localStorage.getItem('breakLog') || '[]');
} catch (e) {
return [];
}
}

function saveLog(log) {
localStorage.setItem('breakLog', JSON.stringify(log));
}

function addLogEntry(entry) {
const log = loadLog();
log.unshift(entry);
saveLog(log);
renderLog();
}

function renderLog() {
const log = loadLog();
if (log.length === 0) {
logContainer.innerHTML = '<div class="empty-log">No breaks recorded yet</div>';
return;
}
let html = '<table><thead><tr><th>Name</th><th>Break</th><th>Actual</th><th>Status</th><th>Time</th></tr></thead><tbody>';
log.forEach(e => {
html += `<tr>
<td>${escapeHtml(e.name)}</td>
<td>${escapeHtml(e.breakLabel)}</td>
<td>${e.actual}</td>
<td class="${e.statusClass}">${e.status}</td>
<td>${e.time}</td>
</tr>`;
});
html += '</tbody></table>';
logContainer.innerHTML = html;
}

function escapeHtml(str) {
const div = document.createElement('div');
div.textContent = str;
return div.innerHTML;
}

clearLogBtn.onclick = () => {
if (confirm('Clear all logged breaks?')) {
saveLog([]);
renderLog();
}
};

breakRow.addEventListener('click', (e) => {
const btn = e.target.closest('.break-btn');
if (btn) selectBreak(btn.dataset.break);
});

nameInput.addEventListener('keydown', e => { if (e.key === 'Enter') startBreak(); });

renderActive();
renderLog();
